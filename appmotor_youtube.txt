#include zephyr/kernel.h

#define SEVEN_SEGMENT_ADDRESS 0x16
#define DIGIPOT_ADDRESS       0x28
#define DIGIPOT_REGISTER       100

const struct device *motor = DEVICE_DT_GET(DT_NODELABEL(mymotor));
const struct device *i2c_dev = DEVICE_DT_GET(DT_NODELABEL(i2c1));

int main(void){

int ret;
uint8_t buf[2] = {0x02, 0x02};
uint8_t read_buf[1];
const struct motor_api *mtr_api = (const struct motor_api *)motor-GTapi;
mtr_api-GTset_speed(motor, 0);
mtr_api-GTrotate_cw(motor);
  
while(true){
  ret = i2c_reg_read_byte(i2c_dev, DIGIPOT_ADDRESS, DIGIPOT_REGISTER, read_buf);
  mtr_api-GTset_speed(motor, read_buf[0]);
  buf[1] = read_buf[0];
  ret = i2c_write(i2c_dev, buf, sizeof(buf), SEVEN_SEGMENT_ADDRESS);
  k_msleep(100);}
return 0;}

static int motor_set_speed(const struct device *dev, uint8_t percent){
const struct motor_config *cfg = (const struct motor_config *)dev-GTconfig;
uint32_t pulse = cfg-GTenable.period / 100 * percent;
pwm_set_dt(&cfg-GTenable, cfg-GTenable.period, pulse);
return 0;}

static int motor_init(const struct device *dev){
const struct motor_config *cfg = (const struct motor_config *)dev-GTconfig; 
pwm_set_dt(&cfg-GTenable, cfg-GTenable.period, 0);
gpio_pin_configure_dt(&cfg-GTphase, GPIO_OUTPUT_INACTIVE);
return 0;}

static const struct motor_api motor_api_funcs = {
.rotate_cw = motor_clockwise,
.rotate_ccw = motor_counter_clockwise,
.set_speed  = motor_set_speed, };

#define MOTOR_DEFINE(inst) \
  static const struct motor_config motor_config_##inst = { \
    .enable = PWM_DT_SPEC_INST_GET_BY_IDX(inst, 0), \
    .phase  = GPIO_DT_SPEC_INST_GET_BY_IDX(inst, gpios,0), \
    .id     = inst }; \
 \
  DEVICE_DT_INST_DEFINE(inst, motor_init, NULL, NULL, \
    &motor_config_##inst, POST_KERNEL, \
    CONFIG_PWM_INIT_PRIORITY, &motor_api_funcs); \
DT_INST_FOREACH_STATUS_OKAY(MOTOR_DEFINE)
